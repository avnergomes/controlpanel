// PROXY - doGet (Control Panel le dados)
// ═══════════════════════════════════════════════════════════════════════════

function doGet(e) {
  var token = (e && e.parameter && e.parameter.token) || '';

  // Se nao tem token, retorna status
  if (!token) {
    return jsonResponse_({
      status: 'ok',
      message: 'DataGeo Unified - Proxy + Tracking LGPD',
      version: '3.1',
      sites: SITES.length,
      trackingFields: TRACKING_COLUMNS.length
    });
  }

  // Verificar token
  if (token !== AUTH_TOKEN) {
    return jsonResponse_({ error: 'unauthorized' });
  }

  var siteFilter = (e && e.parameter && e.parameter.site) || null;

  // Verificar cache
  var cache = CacheService.getScriptCache();
  var cacheKey = 'proxy_' + (siteFilter || 'all');
  var cached = cache.get(cacheKey);

  if (cached) {
    return jsonResponse_(JSON.parse(cached));
  }

  // Buscar dados frescos
  var sitesToFetch = siteFilter
    ? SITES.filter(function (s) { return s.key === siteFilter; })
    : SITES;

  var result = {
    fetchedAt: new Date().toISOString(),
    sites: {},
  };

  for (var i = 0; i < sitesToFetch.length; i++) {
    var site = sitesToFetch[i];
    try {
      var rows = readSiteRows_(site);
      result.sites[site.key] = {
        name: site.name,
        kind: site.kind,
        rows: rows,
        status: 'ok',
      };
    } catch (err) {
      result.sites[site.key] = {
        name: site.name,
        kind: site.kind,
        rows: [],
        status: 'error',
        error: err.message,
      };
    }
  }

  // Cachear
  try {
    var json = JSON.stringify(result);
    if (json.length < 100000) {
      cache.put(cacheKey, json, CACHE_SECONDS);
    }
  } catch (e) {}

  return jsonResponse_(result);
}

function readSiteRows_(site) {
  var allRows = [];
  var ss = SpreadsheetApp.openById(site.sheetId);

  for (var g = 0; g < site.gids.length; g++) {
    var sheet = getSheetByGid_(ss, site.gids[g]);
    if (!sheet) continue;

    var data = sheet.getDataRange().getValues();
    if (data.length < 2) continue;

    var headers = data[0].map(function (h) {
      return String(h).trim();
    });

    for (var r = 1; r < data.length; r++) {
      var row = {};
      for (var c = 0; c < headers.length; c++) {
        if (!headers[c]) continue;
        var val = data[r][c];
        if (val instanceof Date) {
          val = val.toISOString();
        }
        row[headers[c]] = val;
      }
      allRows.push(row);
    }
  }

  return allRows;
}

function getSheetByGid_(spreadsheet, gid) {
  var sheets = spreadsheet.getSheets();
  for (var i = 0; i < sheets.length; i++) {
    if (sheets[i].getSheetId() === gid) {
      return sheets[i];
    }
  }
  return null;
}

// ═══════════════════════════════════════════════════════════════════════════
// TRACKING - doPost (Sites enviam dados de visita)
// ═══════════════════════════════════════════════════════════════════════════

function doPost(e) {
  try {
    var data = validatePayload_(e.postData.contents);
    if (!data) {
      return jsonResponse_({ status: 'error', message: 'invalid payload' });
    }

    var origin = data.origin || '';
    if (!isAllowedOrigin_(origin)) {
      return jsonResponse_({ status: 'error', message: 'forbidden origin' });
    }

    // Rate limit
    var rateLimitKey = (data.timestamp || '') + '_' + origin;
    if (!checkRateLimit_(rateLimitKey)) {
      return jsonResponse_({ status: 'error', message: 'rate limited' });
    }

    // Identificar site
    var site = identifySite_(data);
    if (!site) {
      return jsonResponse_({
        status: 'error',
        message: 'unknown site',
        debug: { origin: origin, page: data.page || '', pathname: data.pathname || '' }
      });
    }

    // Adicionar nome do site
    data.site = site.name;

    // Salvar
    saveTracking_(site, data);

    return jsonResponse_({
      status: 'success',
      site: site.name,
      campos: TRACKING_COLUMNS.length
    });

  } catch (error) {
    return jsonResponse_({
      status: 'error',
      message: error.toString()
    });
  }
}

function identifySite_(data) {
  var page = data.page || data.pathname || '';
  var origin = data.origin || '';

  for (var i = 0; i < SITES.length; i++) {
    var site = SITES[i];
    var urlKey = site.urlKey;

    if (urlKey.indexOf('/') !== -1) {
      // Subpath (ex: avnergomes.github.io/vbp-parana)
      var parts = urlKey.split('/');
      var host = parts[0];
      var path = '/' + parts.slice(1).join('/');
      if ((page.indexOf(host) !== -1 || origin.indexOf(host) !== -1) && page.indexOf(path) !== -1) {
        return site;
      }
    } else {
      // Dominio simples
      if (page.indexOf(urlKey) !== -1 || origin.indexOf(urlKey) !== -1) {
        return site;
      }
    }
  }

  // Fallback: tentar pelo pathname
  if (origin && origin.indexOf('avnergomes.github.io') !== -1) {
    var pathname = data.pathname || '';
    var firstPath = pathname.split('/').filter(Boolean)[0] || '';
    for (var j = 0; j < SITES.length; j++) {
      if (SITES[j].urlKey.indexOf('/' + firstPath) !== -1) {
        return SITES[j];
      }
    }
  }

  return null;
}

function saveTracking_(site, data) {
  var ss = SpreadsheetApp.openById(site.sheetId);
  var sheet = ss.getSheetByName(site.trackingSheet);

  if (!sheet) {
    sheet = ss.insertSheet(site.trackingSheet);
    setupTrackingHeaders_(sheet);
  }

  var rowData = TRACKING_COLUMNS.map(function(col) {
    return sanitize_(data[col]);
  });

  sheet.appendRow(rowData);
}

function setupTrackingHeaders_(sheet) {
  sheet.getRange(1, 1, 1, TRACKING_COLUMNS.length).setValues([TRACKING_COLUMNS]);
  var headerRange = sheet.getRange(1, 1, 1, TRACKING_COLUMNS.length);
  headerRange.setFontWeight('bold');
  headerRange.setBackground('#1a56db');
  headerRange.setFontColor('#ffffff');
  sheet.setFrozenRows(1);
}

// ═══════════════════════════════════════════════════════════════════════════
// SEGURANCA E HELPERS
// ═══════════════════════════════════════════════════════════════════════════

function isAllowedOrigin_(origin) {
  if (!origin) return false;
  for (var i = 0; i < ALLOWED_ORIGINS.length; i++) {
    if (origin === ALLOWED_ORIGINS[i]) return true;
    if (origin.indexOf(ALLOWED_ORIGINS[i]) === 0) return true;
  }
  return false;
}

function checkRateLimit_(key) {
  if (!key) return true;
  var cache = CacheService.getScriptCache();
  var cacheKey = 'rl_' + key.substring(0, 30);
  var current = cache.get(cacheKey);
  var count = current ? parseInt(current, 10) : 0;
  if (count >= 60) return false;
  cache.put(cacheKey, String(count + 1), 60);
  return true;
}

function validatePayload_(raw) {
  if (!raw || raw.length > 5000) return null;
  try {
    var data = JSON.parse(raw);
    if (typeof data !== 'object' || data === null) return null;
    return data;
  } catch (e) {
    return null;
  }
}

function sanitize_(value) {
  if (value === null || value === undefined) return '';
  var str = String(value).substring(0, 500);
  return str.replace(/[<>\"\'\\]/g, '');
}

function jsonResponse_(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

// SETUP (executar UMA VEZ para criar abas de tracking)
// ═══════════════════════════════════════════════════════════════════════════

function setupAllSheets() {
  var results = [];

  for (var i = 0; i < SITES.length; i++) {
    var site = SITES[i];

    // Skip sites com sheetId placeholder
    if (site.sheetId.indexOf('CRIAR_PLANILHA') !== -1) {
      results.push({ site: site.name, status: 'skipped', message: 'sheetId nao configurado' });
      continue;
    }

    try {
      var ss = SpreadsheetApp.openById(site.sheetId);
      var existing = ss.getSheetByName(site.trackingSheet);

      if (existing) {
        existing.getRange(1, 1, 1, TRACKING_COLUMNS.length).setValues([TRACKING_COLUMNS]);
        var headerRange = existing.getRange(1, 1, 1, TRACKING_COLUMNS.length);
        headerRange.setFontWeight('bold');
        headerRange.setBackground('#1a56db');
        headerRange.setFontColor('#ffffff');
        results.push({ site: site.name, status: 'updated', rows: existing.getLastRow() - 1 });
      } else {
        var sheet = ss.insertSheet(site.trackingSheet);
        setupTrackingHeaders_(sheet);
        results.push({ site: site.name, status: 'created' });
      }
    } catch (error) {
      results.push({ site: site.name, status: 'error', message: error.toString() });
    }
  }

  Logger.log('═══════════════════════════════════════════');
  Logger.log('SETUP COMPLETO - DataGeo Unified v3.1');
  Logger.log('═══════════════════════════════════════════');
  Logger.log('Sites configurados: ' + SITES.length);
  Logger.log('');
  results.forEach(function(r) {
    Logger.log('  ' + r.site + ': ' + r.status + (r.rows ? ' (' + r.rows + ' registros)' : '') + (r.message ? ' - ' + r.message : ''));
  });
  Logger.log('');
  Logger.log('IMPORTANTE: Lembre-se de:');
  Logger.log('1. Gerar novo AUTH_TOKEN (use crypto.randomUUID() no console do navegador)');
  Logger.log('2. Criar planilhas para sites com sheetId placeholder');
  Logger.log('3. Fazer novo deployment e atualizar secrets no GitHub');

  return results;
}
